#include <iostream>
#include<string>
#include <algorithm>

using namespace std;

int main()
{
    string names[] ={
        "Hadeer Mahmoud" ,
        "Youssef Ail" ,
        "Ahmed Mostafa" ,
    } ;
    int n=3 ;
    sort(names , names + n) ;

    // sort names in ascending order sort (names , names + n) ;
    cout << "Students names in Ascending Order :\n" ;
    for (int i= 0 ; i<n ; i++){
        cout << names[i] << endl ;
    }


    return 0;
}
