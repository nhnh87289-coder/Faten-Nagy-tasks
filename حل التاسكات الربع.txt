#include <iostream>
#include <fstream>
#include <vector>
#include <string>
using namespace std;

struct Person
{
    string name;
    int age;
};

int main()
{
    string names[4] = {
        "Simon",
        "Marina",
        "Martina",
        "Rania"
    };


    ofstream outFile("persons.txt");

    if (!outFile)
    {
        cout << "Error opening file!";
        return 1;
    }

    cout << "Enter persons ages:" << endl;

    for (int i = 0; i < 4; i++)
    {
        int age;

        cout << names[i] << " age: ";
        cin >> age;

        outFile << names[i] << " " << age << endl;
    }

    outFile.close();


    vector<Person> people;

    ifstream inFile("persons.txt");

    Person p;

    while (inFile >> p.name >> p.age)
    {
        people.push_back(p);
    }

    inFile.close();

    if (people.empty())
    {
        cout << "No data found!";
        return 1;
    }


    int youngest = people[0].age;
    int oldest = people[0].age;

    for (int i = 1; i < people.size(); i++)
    {
        if (people[i].age < youngest)
            youngest = people[i].age;

        if (people[i].age > oldest)
            oldest = people[i].age;
    }


    ofstream resultFile("results.txt");

    resultFile << "Youngest person(s):" << endl;

    for (int i = 0; i < people.size(); i++)
    {
        if (people[i].age == youngest)
            resultFile << people[i].name
                       << " - " << people[i].age << endl;
    }

    resultFile << endl << "Oldest person(s):" << endl;

    for (int i = 0; i < people.size(); i++)
    {
        if (people[i].age == oldest)
            resultFile << people[i].name
                       << " - " << people[i].age << endl;
    }

    resultFile.close();


    cout << endl << "All Persons:" << endl;

    for (int i = 0; i < people.size(); i++)
    {
        cout << people[i].name
             << " - " << people[i].age << endl;
    }

    cout << endl << "Youngest age: " << youngest << endl;
    cout << "Oldest age: " << oldest << endl;

    cout << "Results saved successfully!" << endl;

    return 0;
